60inline void writeBatchModeNormal(const std::string &filename, const int source, const int dest,
61 const std::vector<int> &BestDrivingRoute,
62 const std::vector<int> &AlternativeDrivingRoute, const int best_time,
63 const int alternative_time = 0) {
64 std::ofstream file(filename); //declare file as a output file
65 if (!file.is_open() || file.fail()) {
66 std::cout << "Error opening the file!" << std::endl;
67 exit(0);
68 }
69
70 file << "Source:" << source << std::endl;
71 file << "Destination:" << dest << std::endl;
72 file << "BestDrivingRoute:";
73 if (BestDrivingRoute.empty()) {
74 //no path obtained
