26template<class T>
27bool walkingrelax(Edge<T> *edge) {
28 // d[u] + w(u,v) < d[v]
29 if (edge->getOrig()->getWalkingDist() + edge->getWalkingWeight() < edge->getDest()->getWalkingDist()) {
30 // we have found a better way to reach v
31 edge->getDest()->setWalkingDist(edge->getOrig()->getWalkingDist() + edge->getWalkingWeight());
32 // d[v] = d[u] + w(u,v)
33 edge->getDest()->setPath(edge); // set the predecessor of v to u; in this case the edge from u to v
34 return true;
35 }
36 return false;
37}

49void walkingdijkstra(Graph<T> *g, const int &origin, vector<int> *rn, vector<pair<int, int> > *re) {
51 for (auto v: g->getVertexSet()) {
52 v->setWalkingDist(INF);
53 v->setPath(nullptr);
54 v->setVisited(false);
55 }
57 if (rn != nullptr) {
58 for (int n: *rn) {
59 auto v = g->findVertex(n);
60 if (v != nullptr) {
61 v->setVisited(true);
62 }
63 }
64 }
66 if (re != nullptr) {
67 for (pair<int, int> en: *re) {
68 auto v = g->findVertex(en.first);
69 if (v != nullptr) {
70 for (auto e: v->getAdj()) {
71 if (e->getDest() == g->findVertex(en.second)) {
72 e->setWalkingWeight(INF);
73 }
74 }
75 }
76 }
77 }
78 auto s = g->findVertex(origin);
79 if (s == nullptr) {
80 return;
81 }
82 s->setWalkingDist(0);
83
84 MutablePriorityQueue<Vertex<T> > q;
85 q.insert(s);
86 while (!q.empty()) {
87 auto v = q.extractMin();
88 v->setVisited(true);
89 for (auto e: v->getAdj()) {
90 auto oldDist = e->getDest()->getWalkingDist();
91 if (relax(e) && e->getDest()->isVisited() == false) {
92 if (oldDist == INF) {
93 q.insert(e->getDest());
94 } else {
95 q.decreaseKey(e->getDest());
96 }
97 }
98 }
99 }
100}

109template<class T>
110static std::vector<T> getWalkingPath(Graph<T> *g, const int &origin, const int &dest) {
111 std::vector<T> res;
112 auto v = g->findVertex(dest);
113 if (v == nullptr || v->getWalkingDist() == INF) {
114 // missing or disconnected
115 return res;
116 }
117 res.push_back(v->getId());
118 while (v->getPath() != nullptr) {
119 v = v->getPath()->getOrig();
120 res.push_back(v->getId());
121 }
122 reverse(res.begin(), res.end());
123 if (res.empty() || res[0] != origin) {
124 //std::cout << "Origin not found!!" << std::endl;
125 //return empty vector
126 std::vector<T> empty;
127 return empty;
128 }
129 return res;
130}
