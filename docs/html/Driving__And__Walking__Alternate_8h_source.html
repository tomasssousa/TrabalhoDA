28template<class T>
29bool walkingrelax(Edge<T> *edge) {
30 // d[u] + w(u,v) < d[v]
31 if (edge->getOrig()->getWalkingDist() + edge->getWalkingWeight() < edge->getDest()->getWalkingDist()) {
32 // we have found a better way to reach v
33 edge->getDest()->setWalkingDist(edge->getOrig()->getWalkingDist() + edge->getWalkingWeight());
34 // d[v] = d[u] + w(u,v)
35 edge->getDest()->setPath(edge); // set the predecessor of v to u; in this case the edge from u to v
36 return true;
37 }
38 return false;
39}

50template<class T>
51void walkingdijkstra(Graph<T> *g, const int &origin, vector<int> *rn, vector<pair<int, int> > *re) {
53 for (auto v: g->getVertexSet()) {
54 v->setWalkingDist(INF);
55 v->setPath(nullptr);
56 v->setVisited(false);
57 }
59 if (rn != nullptr) {
60 for (int n: *rn) {
61 auto v = g->findVertex(n);
62 if (v != nullptr) {
63 v->setVisited(true);
64 }
65 }
66 }
68 if (re != nullptr) {
69 for (pair<int, int> en: *re) {
70 auto v = g->findVertex(en.first);
71 if (v != nullptr) {
72 for (auto e: v->getAdj()) {
73 if (e->getDest() == g->findVertex(en.second)) {
74 e->setWalkingWeight(INF);
75 }
76 }
77 }
78 }
79 }
80 auto s = g->findVertex(origin);
81 if (s == nullptr) {
82 return;
83 }
84 s->setWalkingDist(0);
85
86 MutablePriorityQueue<Vertex<T> > q;
87 q.insert(s);
88 while (!q.empty()) {
89 auto v = q.extractMin();
90 v->setVisited(true);
91 for (auto e: v->getAdj()) {
92 auto oldDist = e->getDest()->getWalkingDist();
93 if (relax(e) && e->getDest()->isVisited() == false) {
94 if (oldDist == INF) {
95 q.insert(e->getDest());
96 } else {
97 q.decreaseKey(e->getDest());
98 }
99 }
100 }
101 }
102}

111template<class T>
112static std::vector<T> getWalkingPath(Graph<T> *g, const int &origin, const int &dest) {
113 std::vector<T> res;
114 auto v = g->findVertex(dest);
115 if (v == nullptr || v->getWalkingDist() == INF) {
116 // missing or disconnected
117 return res;
118 }
119 res.push_back(v->getId());
120 while (v->getPath() != nullptr) {
121 v = v->getPath()->getOrig();
122 res.push_back(v->getId());
123 }
124 reverse(res.begin(), res.end());
125 if (res.empty() || res[0] != origin) {
126 //std::cout << "Origin not found!!" << std::endl;
127 //return empty vector
128 std::vector<T> empty;
129 return empty;
130 }
131 return res;
132}

144template<class T>
145vector<int> findValidParkingNodes(Graph<T> *g, int origin, int destination, int maxWalkTime) {
146 vector<int> parkingNodes;
147 for (auto v: g->getVertexSet()) {
148 if (v->getParking() == 1 && v->getId() != origin && v->getId() != destination) {
149 walkingdijkstra(g, v->getId(), {}, {});
150 if (g->findVertex(destination)->getWalkingDist() <= maxWalkTime) {
151 parkingNodes.push_back(v->getId());
152 }
153 }
154 }
155 return parkingNodes;
